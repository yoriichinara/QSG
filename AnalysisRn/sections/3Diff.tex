\section{O Diferencial} 

\renewcommand{\hom}{\operatorname{Hom}}
\begin{definition}
    Sejam \(f:U\ab\subseteq \R^n \to \R\) e \(p\in U\). Dizemos que \(f \) é \emph{diff} em \(p\) sse \(\exists \ell \in (\R^n)'\) tal que \(f(p+v) = f(p) + \ell\cdot v + o\|v\|, \text{ quando} \ v\to 0. \footnote{\((\R^n)' := \hom_\R(\R^n, \R)\), o espaço dual, olha as notas de Álgebra Linear II.}\)
\end{definition}
%\vspace{-0.4cm}
\begin{note}
    Se \(f\) é diff em \(p\), então \(\forall \vec{v}\in \R^n\), \( \frac{\partial f }{\partial v}(p) = \ell \cdot v \).% \textcolor{gray}{\(\rightarrow \) Vá de uma definição a outra} %\todo[orange, noline]{La derivada \(v\)-direccional es lineal}%En particular \direccionales son lineales en esa dirección.}. \footnote{Esto también implica que las derivadas direccionales son lineales en esa dirección.}
\end{note}

\begin{definition}
    Seja \(f\) diff em \(p\). O \emph{diferencial} de \(f\) em \(p\) é a função linear \(df(p):\V_p\to \R\) dada por 
    \(df(p)\cdot v := \sum^n f_{x_j}(p)\cdot v_j\).
\end{definition} %\section*{Aula III}

\alerta{\centering \(\forall j\leq n, \ \exists f_{x_j}(p)\) \textcolor{red}{\(\nRightarrow \)} \(df(p)\) linear \textcolor{red}{\(\nRightarrow\)} \(f\) diff em \(p\)}
\begin{example}
    \(f: \R^n \ni \textbf{x} \mapsto \|\textbf{x}\|^2 \in \R\) e diff em \(\R^n\). \textcolor{gray}{\(\rightarrow\) Como em nosso primeiro exemplo}
\end{example}

\begin{proposition}
    Se \(f\) é diff em \(p\), então é contínua em \(p\). \textcolor{gray}{\(\rightarrow \) Faz \(v\to 0\) na definição }
\end{proposition}

  
\ejemplo{\centering \(\nexists df(p) \ \textcolor{verdeoscuro}{\Leftarrow} \ (\R^n)' \not\ni\exists \frac{\partial f}{\partial v}(p) \ \  \textcolor{verdeoscuro}{\big|} \ \ \forall \vec{v} \in \R^n,\  \exists \frac{\partial f }{\partial v }(p)\) \textcolor{red}{\(\nRightarrow\)} \(\exists df(p)\).
\tcblower

\end{example}

\begin{definition}
    Se \(f\) é diff em \(p\), então chamamos de \emph{gradiente} de \(f\) em \(p\) o vetor 
    \[\nabla f(p) := \left(\frac{\partial f}{\partial x_1}(p),\ldots,\frac{\partial f}{\partial x_n}(p)\right)^t. \]
\end{definition}

\begin{note}
    \(\forall \vec{v}\in \R^n,\ \langle \nabla f(p) , v\rangle = \frac{\partial f}{\partial v}(p)\). Se \(\|v\|=1\), então \( |\langle \nabla f (p), v\rangle| \leq \|\nabla f(p)\|\), o gradiente aponta na direção de maior crescimento.  
\end{note}
