\section{Formula de Taylor}

\ejemplo{
    A gente quer aproximar funções por polinômios da forma \(f(p+v) = P_k[v] + \Gamma_k[v]\), polinômio ordem \(k\) mas erro ordem \(\sigma (\|v\|^k)\).
    \tcblower
    \begin{example}
        Se \(f\in C^2(U)\) e \(p\in U\), então  
        \[\textstyle  f(p+v) = \underbrace{f(p) + \sum \frac{\partial f }{\partial x_j}(p)v_j+ \frac{1}{2}\sum \frac{\partial f}{\partial x_i \partial x_j}(p) v_iv_j}_{P_2[v]} + \underbrace{\sigma(\|v\|^2)}_{\Gamma_2[v]}.\]
    \end{example}
}

\begin{definition}
    Seja \(k\geq 2\). Dizemos que \(f:U\ab\subseteq \R^n \to \R\) é \(k\)\emph{-diff} em \(p\in U\) se \(f\) é diff em \(p\) e \(\exists \V_p\) tal que \(\forall j\leq n, \ \frac{\partial f }{\partial x_j}: \V_p \to \R\) é \((k-1)\)-diff em \(p\). 
\end{definition}

\Ei 

\begin{example}
    Se \(f \in C^k(U)\) então \(\forall p\in U\), \(f\) é \(k\)-diff em \(p\). 
\end{example} 

\begin{definition}
    Seja \(f\) função \(k\)-diff em \(p\), o \emph{diferencial \(k\)-ésimo} de \(f\) em \(p\) é o dado por %\(d^kf(p): \R^n \to \R\) tal que 
    \[v\mapsto d^{k}f(p)v^{\otimes k} = \sum_{k=|\alpha |}\frac{k!}{\alpha!}\partial^\alpha f (p) v^\alpha. \footnote{Entender \(\alpha ! = \alpha_1! \alpha_2! \cdots \alpha_k!\).}\] 

\end{definition}

%    \(d^kf(p)v^{\otimes k}\) é um polinômio homogêneo de grau \(k\)
%\end{note}

\ejemplo{\centering 
    \(d^2f(p)v = vA(p) v^T \) é forma bilinear quadrâtica de grau 2  
    \tcblower
    \begin{example}
    Para \(f\in C^2(\R^2)\), sendo \(v=(h,k)\) temos \[\textstyle d^2f(p)v^{\otimes 2} = \frac{\partial^2 f}{\partial x^2}(p)h^2 + 2 \frac{\partial^2 f }{\partial x\partial y}(p)hk+ \frac{\partial^2 f}{\partial y^2}(p)k^2.\]  
    %\- \vspace{-0.5cm}
    %\textcolor{gray}{\footnotesize \[\ \hspace{2.8cm} \alpha = (2,0) \hspace{1.5cm} \alpha = (1,1) \hspace{1.5cm} \alpha = (0,2)\]}
        \comentario{\( (1,1) = \alpha \leadsto |\alpha| = 1+1 = 2 \leadsto  \frac{k!}{\alpha_1!\alpha_2!} = \frac{2!}{1!1!} =2\)}
    \end{example}
}
    
\Ef

\begin{definition}
    Seja \(f\) função \(2\)-diff  em \(p\), a \emph{Hessiana} da \(f\) em \(p\), é a matriz   
    \[ Hf(p) = \left(\frac{\partial^2 f}{\partial x_i\partial x_j}(p)\right)_{i,j} =  \begin{bmatrix}
        \frac{\partial^2 f }{\partial x_1^2}(p) & \cdots &  \frac{\partial^2 f }{\partial x_1 \partial x_n}(p) \\
        \rotatebox{90}{$\cdots$} & \rotatebox{45}{$\cdots$} & \rotatebox{90}{$\cdots$} \\
        \frac{\partial^2 f }{\partial x_n \partial x_1}(p) & \cdots & \frac{\partial^2 f }{\partial x_n^2}(p)
    \end{bmatrix}.\]
\end{definition}
\begin{note}
    Pelo \(\blacksquare\) \hyperlink{thmSchw}{(\emph{Schwarz})}, \(\exists !Hf(p) \in \operatorname{Sym}_n(\R)\) \(: d^2f(p)v^{\otimes 2} = \langle Hf(p)\cdot v, v\rangle \). 
\end{note}

\begin{definition}
    \(f \in C^{k} (\V_p)\) se anula à ordem \(k+1\) em \(p\) se \(\forall \alpha :  |\alpha|\leq k\), \( d^\alpha f (p)=0\). 
\end{definition}

\Ei

\begin{exercise}
    Se \(f(\textbf{x})= \sum_{|\alpha|\leq k} c_\alpha \textbf{x}^\alpha \) se anula identicamente numa vizinhaza do origem então \(\forall \alpha, \ c_\alpha = 0\). \textcolor{gray}{\(\rightarrow \) Hmmm }
\end{exercise}

\hypertarget{thm51}{}
\begin{theorem}
    \textbf{5.1.} Sejam \(k\geq 1\) e \(f\) uma função \(k\)-diff em \(0\in \R^n\). Se \(f\) se anula à ordem \(k+1 \) então \(f(v)=\sigma(\|v\|^k)\).
\end{theorem}
\demo{Faz por indução, no paso indutivo use o \(\blacksquare\) \hyperlink{thmTVMRn}{(\emph{TVM - \(\R^n\)})} para representar \(f(v)\) e calcule \(\lim\limits_{v\to 0} \frac{f(v)}{\|v\|^k} \). Considere a função  
\- \vspace{-0.3cm}\[\textstyle \Gamma_k(v) : v \mapsto f(p+v) - \sum^k \frac{1}{j!} \  d^jf(p) v^{\otimes j}, \]
observe que ela é \(k\)-diff no \(0\) e se anula à ordem \(k+1\).} 

\begin{corollary}[]%[Colorario]
\hypertarget{coro52}{}
    \textbf{5.2.} (\emph{Fórmula de Taylor Infinitesimal}) Se \(f\) é \(k\)-diff em \(p\) e fazendo \(\textbf{x}=p+v\) no contexto da demostração do \(\blacksquare\) \hyperlink{thm51}{5.1.} temos,
    \[f(\textbf{x}) = \sum_{|\alpha|\leq k}\frac{\partial^\alpha f (p)}{\alpha !} \ (\textbf{x}-p)^\alpha + \sigma(\|\textbf{x}-p\|^k). \]
    %\[f(p+v)= \sum^k \frac{1}{j!} \  d^jf(p) v^{\otimes j}  + \sigma(\|v\|^k).\] 
\end{corollary}

\begin{proposition}[\emph{Taylor com Resto de Lagrange - \(\R\)}]
    Sejam \(a \in I\ab\subset \R\) e \(\varphi \in C^{k+1}(I)\), então \(\varphi (x) = P_k[x] + \Gamma_k[x]\) e \( \exists c \in I\), tal que \(\Gamma_k(x) = \frac{\varphi^{(k+1)}(c)}{(k+1)!} \ (x-a)^{(k+1)}\). 
%    \begin{enumerate}[label = \roman*.]

%    \end{enumerate} 
\end{proposition}

\begin{corollary}[\emph{Taylor com Resto de Lagrange - \(\R^n\)} ]
    Seja \(f\in C^{k+1}(U)\). Então, \(\forall p\in U\) e \(\vec{v}\in \R^n\) tais que \([p,p+tv]\subseteq U\) temos \(f(p+v) = P_k[v] + \Gamma_k[v]\) e \(\exists \theta \in (0,1)\) tal que \(\Gamma_k[v] = \frac{d^{k+1}f(p+\theta v)}{(k+1)!}  v^{\otimes (k+1)}\). 
\end{corollary}
\demo{Aplique o sabido numa variável à função \(\varphi(t) = f(p+tv)\). Expresse o diferencial como soma, e desenvolva até chegar à forma descrita. }
