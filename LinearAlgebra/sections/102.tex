\subsection{Produto Tensorial}

\begin{definition}
    Seja \((V_j)\) família de \(\F\)-espaços vetoriais, \(j\leq k \in \N\). Um \(\F\)\emph{-produto tensorial} pra \((V_j)\) é um par universal \((\phi, V)\) sobre \(X = \Pi V_j\) onde,   
    \begin{itemize}[left = 1cm]
        \item \(V\) é \(\F\)-espaço vetorial e \(\phi \in \hom^k(\Pi V_j, V)\). 
        \item \(P_1 = \) "ser $k$-linear" e \(P_2 = \) "ser linear".
    \end{itemize}  
\end{definition}

\hypertarget{thm1021}{}
\theoremnum{10.2.1.}
\begin{theorem}
    \label{thm:1021}
    \(\forall (V_j),\ \exists (\phi, V)\) produto tensorial. Além disso, se \(\forall j\leq k, \ \alpha_j\) fora base de \(V_j\), então \(\phi \circ (\Pi \alpha_j)\) é base de \(V\).  
\end{theorem}

\demo{Sejam \(\alpha := \Pi \alpha_j\), \(V : \dim V = |\alpha|\) e \(\iota : \alpha \to V\) base pra \(V\). Segue do \(\blacksquare \) \ref{thm:9.1.3} que \(\exists ! \phi \in \hom^k(\Pi V_j, V): \phi|_\alpha = \iota\). Conclui mostrando que \((\phi, V)\) é universal.} 

\theoremnum{10.2.2.}
\begin{theorem}
    \label{thm:1022}
    Seja \((\phi, V)\) produto tensorial pra \((V_j)\). Então, \(\forall W\) \(\F\)-espaço vetorial a função \(\Psi: \hom^k(\Pi V_j, W) \ni \psi \mapsto \widetilde{\psi} \in \hom(V,W)\) é isomorfismo. 
\end{theorem}

\demo{Injetora por conseqüência do {\scriptsize \(\square\)} \hyperlink{lemma1012}{10.1.2.} e sobrejetiva pois se \(T \in \hom(V,W)\), então \(\psi = T \circ \phi \in  \hom^k(\Pi V_j, W)\). Pra ver a linearidade mostra que \(\forall \psi, \xi \in \hom^k(\Pi V_j, W)\) e \(\forall \lambda \in \F\), \((\widetilde{\psi}+ \lambda \widetilde{\xi}) \circ \phi = \psi + \lambda \xi\). } 

\begin{note}
    Notação padrão pra producto tensorial, \(\textstyle (\phi, V) = \left(\otimes, \otimes V_j\right)\). A $k$-linearidade fica,  \( \cdots \otimes v_{1,j_0} + \lambda v_{2,j_0} \otimes \cdots = \cdots \otimes v_{1,j_0} \otimes \cdots + \lambda \cdots \otimes v_{2,j_0}\otimes \cdots \).
    %O corpo e \(k\) apareceram só se precisar.
%    \[\otimes ((v_{1,j})_{j_0} + \lambda(v_{2,j})_{j_0}) = \otimes (v_{1,j})_{j_0} + \lambda \otimes (v_{2,j})_{j_0}.\]
\end{note}

\propositionnum{10.2.3.}
\begin{proposition}
    \label{prop:1023}
    Dado \(1\leq l < k\), \(\exists ! \Upsilon  : \otimes V_j \to (\otimes^l V_j) \otimes (\otimes_{l+1}^k  V_j)\)  isomorfismo tal que, \(\forall j\leq k\), \(\forall v_j \in V_j\), \(\Upsilon (\otimes v_j )= (\otimes^l v_j) \otimes (\otimes_{l+1}^kv_j)\). 
\end{proposition}

\demo{Defina \(\phi_b : \Pi V_j \to W =  (\otimes^l V_j) \otimes (\otimes^k_{l+1} V_j)\) e tente mostrar que o par \((\phi_b, W)\) também é universal em \(\Pi V_j\). O resultado segue do {\scriptsize\(\square\) }\hyperlink{lemma1013}{10.1.3.}.}

\propositionnum{10.2.4.}
\begin{proposition}
    \(\forall j\leq k\) sejam \(W_j \leq V_j\). Então, \(\exists ! \iota : \otimes W_j \hookrightarrow \otimes V_j\) linear injetora tal que \(\forall j\leq k, \ \forall w_j \in W_j, \ \iota (\otimes w_j) = \otimes w_j\). 
\end{proposition}

\demo{No contexto da \(\square\) \ref{prop:1023} \(\exists ! \Upsilon_{_V} :\otimes V_j \to \otimes (V_j)\) e \(\exists !\Upsilon_{_W}: \otimes W_j \to \otimes (W_j)\) isomorfismos. Faça \(\iota = \Upsilon^{-1}_{_V}\circ \Upsilon_{_W} \). }

\begin{corollary}
    \(\otimes v_j = 0 \leftrightharpoons \exists j\leq k : v_j = 0 \). 
\end{corollary}

\begin{note}
    Do \(\blacksquare\) \ref{thm:1021} sabemos que \([\otimes v_j] = \otimes V_j\), então \(\forall \textbf{v} \in \otimes V_j,\ \exists m \in \mathbb{Z}_{\geq 0}\) e \( \exists v_{i,j} \in V_j \setminus\{0\}: \textbf{v} = \sum_{i\leq m} \otimes v_{i,j} \), ou seja, \textbf{v} é soma de \emph{tensores puros}.   
\end{note}

\begin{definition}
    \centering
    \(\pt\textbf{v} := \min\{m \in \mathbb{Z}_{\geq 0} : \textbf{v} = \sum_{i\leq m} \otimes v_{i,j}\}\). 
\end{definition}

\hypertarget{lemma1026}{}
\begin{lemma}
    \textbf{10.2.6.} Seja \(\tu \in V\otimes W : \pt \tu = m\). Se \(\alpha = (v_j)\) e \(\beta = (w_j)\), \(j\leq m\), são famílias de vetores em \(V\) e \( W\) tais que \(\tu = \sum^m v_j \otimes w_j\), então \(\alpha\) e \(\beta \) são l.i..
\end{lemma}

\demo{Faz por contrapositiva, suponha que \(w_m = \sum^{m-1} a_jw_j\) e usa a $k$-linearidade pra ver que \(\tu = \sum^{m-1} (v_j + a_jv_m) \otimes w_j\).}

\begin{corollary}
    \(\forall \tu \in V\otimes W\), \(\pt \tu \leq \min \{\dim V, \dim W\}\). 
\end{corollary}

\hypertarget{lemma1027}{}
\begin{lemma}
    \small  \textbf{10.2.7.} Sejam \(m, p \in \mathbb{Z}_{\geq 0}\), \(\alpha_1 = (v_{1,j})_{j\leq m}\), \(\alpha_2 = (v_{2,j})_{j\leq p}\), famílias em \(V\) e \(\beta_1 = (w_{1,j})_{j\leq m}\), \(\beta_2= (w_{2,j})_{j\leq p}\) famílias em \(W\) tais que \(\sum^m v_{1,j} \otimes w_{1,j} = \sum^p v_{2,j} \otimes w_{2,j}\). Se \(\alpha_1, \alpha_2\) e \(\beta_2\) foram l.i., então \([\beta_1 ] \subseteq [\beta_2]\). Em particular, se \(p = 0\), \(\forall j\leq m\), \(w_j = 0 \).    
\end{lemma}

\propositionnum{10.2.10.}
\begin{proposition}
    \label{prop:10210}
    \(\exists ! \Gamma : V' \otimes W \ni  \textcolor{red}{f \otimes w \mapsto} \ \overset{\underset{\rotatebox{90}{$\in$}}{\hom(V,W)}}{\textcolor{red}{\Gamma(f\otimes w)}}: \overset{\underset{\rotatebox{90}{$\in$}}{V}}{v}\mapsto f(v)w \in W\). \footnote{Propriedade universal no pareamento bilinear \(\psi : V' \times W \ni (f,w) \mapsto f(v)w\).} Além disso, 
\end{proposition}

\begin{enumerate}[label = , left = 0.5cm ]
    \item \vspace{-1.3cm} \begin{exercise}
        \textbf{9.2.5.} Sejam \(m \in \mathbb{Z}_{>0}\) e \((f_j)_{j\leq m}\) família l.i. em \(V'\). Então, \(\forall j\leq m, \exists v \in V : \forall i\leq m, f_i(v) = \delta_{ij}\). \comentario{\(T: V \ni v \twoheadrightarrow (f_j(v)) \in \F^m \ \leftrightharpoons \ T' : (\F^m)' \hookrightarrow V'\)}
    \end{exercise}
    \begin{tcolorbox}[colframe=gray, colback=white, boxrule=0.8pt]
        \centering (a) \(\Gamma \) é injetora 
        \tcblower
        \textcolor{gray}{
            Sejam \(\tu \in \ker \Gamma \),  \((f_j)\) e \( (w_j)\) l.i. em \(V'\) e \(W\) tais que \(\tu = \sum^m f_j \otimes w_j\), se \(m>0\), então do exercício acima \(\exists v: 0 = \Gamma (\tu)(v) = \sum^m f_j(v)w_j \ \rightarrow \leftarrow\).
        }
    \end{tcolorbox}
    \item \begin{tcolorbox}[colframe=gray, colback=white, boxrule=0.8pt]
        \centering (b) \(\forall \tu \in V' \otimes W, \ \pt (\Gamma(\tu)) = \pt \tu\)
        \tcblower
        \textcolor{gray}{
            De novo sejam \(\tu = \sum^m f_j \otimes w_j\), sendo \(\beta = (w_j)\) mostra que \(\operatorname{Im(\Gamma(\tu)) } = [\beta]\), usa de novo o mesmo exercício. 
        }
    \end{tcolorbox}
    \item \begin{tcolorbox}[colframe=gray, colback=white, boxrule=0.8pt]
        \centering (c) \(T \in \operatorname{Im}\Gamma \ \leftrightharpoons \ \pt T < \infty\)
        \tcblower
        \textcolor{gray}{
            (\(\Rightarrow \)) Segue de (b) lembrando que \(\forall \tu \in V\otimes W, \ \pt \tu < \infty\). (\(\Leftarrow\)) Seja \((w_j)_{j\leq m} \) base pra \(\operatorname{Im}T\) e \((v_i)\) base de \(V\), então \(T(v_i) = \sum^m a_{ij} w_j\), \(\forall j\leq m, \exists !f_j: f_j(v_i) = a_{ij} \), cálcula \(\Gamma\left(\sum^m f_j \otimes w_j\right)\) e conclui.  
        }
    \end{tcolorbox}
    \item \begin{tcolorbox}[colframe=gray, colback=white, boxrule=0.8pt]
        \centering (d) \(\Gamma \) sobrejetora  \(\leftrightharpoons \)  \(\dim V  < \infty\) ou \(\dim W < \infty \)
        \tcblower
        \textcolor{gray}{
            (\(\Leftarrow \)) Direta de (c). (\(\Rightarrow\)) Faz por contrapositiva, se \(\dim V \) e \(\dim W\) foram ambas infinitas, \(\exists T \in \hom(V,W): \pt T = \infty\), logo usa (c) de novo.   
        }
    \end{tcolorbox}
\end{enumerate}

\begin{note}
    \(\Gamma: V' \otimes W \cong F := \{T \in \hom(V,W): \pt T < \infty\}\). Se \(\alpha = (v_j)_{j\leq n}\), \(\alpha' = (f_j)\)  e \(\beta = (w_i)_{i\leq m}\), então \([\Gamma(f_j\otimes w_i)]_\beta^\alpha = E_{ij}\). Assim,  
    \[[T]^\alpha_\beta = (a_{ij}) \ \ \Rightarrow  \ \ T = \Gamma \left(\sum_{i,j} a_{ij} f_j \otimes w_i\right). \]
\end{note}

\propositionnum{10.2.11.}
\begin{proposition}
    \(\exists ! \tau \in (V'\otimes V)' \) tal que \(\tau (f\otimes v) = f(v)\). Além disso, se \(\dim V < \infty \), segue da \(\square\) \ref{prop:10210} que \(\forall T \in \ehom (V), \ \tau(\Gamma^{-1}(T)) = \operatorname{tr}T\).  
\end{proposition}

\demo{\(\exists !\tau\) segue da propriedade universal em \(\psi = \operatorname{av}\). Seja \(\alpha = (v_j)\) base pra \(V\), da nota prévia \([T]^\alpha_\alpha \Rightarrow \Gamma^{-1}(T) =  \sum_{i,j} a_{ij}f_j \otimes v_i\), faça a continha \(\tau(\Gamma^{-1}(T))\). }
