\subsection{Produto Tensorial}

\begin{definition}
    Seja \((V_j)\) familia de \(\F\)-espaços vetorias, \(j\leq k \in \N\). Um \(\F\)\emph{-produto tensorial} pra \((V_j)\) é um par universal \((\phi, V)\) sobre \(X = \Pi V_j\) onde,   
    \begin{itemize}[left = 1cm]
        \item \(V\) é \(\F\)-espaço vetorial e \(\phi \in \hom^k(\Pi V_j, V)\). 
        \item \(P_1 = \) "ser $k$-linear" e \(P_2 = \) "ser linear".
    \end{itemize}  
\end{definition}

\hypertarget{thm1021}{}
\theoremnum{10.2.1.}
\begin{theorem}
    \label{thm:1021}
    \(\forall (V_j),\ \exists (\phi, V)\) \(\F\)-produto tensorial. Além disso, se \(\forall j\leq k, \ \alpha_j\) fora base de \(V_j\), então \(\phi \circ (\Pi \alpha_j)\) é base de \(V\).  
\end{theorem}

\demo{Sejam \(\alpha := \Pi \alpha_j\), \(V : \dim V = \# \alpha\) e \(\iota : \alpha \to V\) base pra \(V\) indexada por \(\alpha\). Segue do \(\blacksquare \) \ref{thm:9.1.3} que \(\exists ! \phi \in \hom^k(\Pi V_j, V): \phi|_\alpha = \iota\). Conclui mostrando que o par \((\phi, V)\) é universal.} 

\theoremnum{10.2.2.}
\begin{theorem}
    Seja \((\phi, V)\) um produto tensorial pra \((V_j)\). Então, \(\forall W\) \(\F\)-espaço vetorial a função \(\Gamma: \hom^k(\Pi V_j, W) \to \hom(V,W)\) é isomorfismo. 
\end{theorem}

\demo{Injetora por conseqüência do {\scriptsize \(\square\)} \hyperlink{lemma1012}{10.1.2.} e sobrejetiva pois se \(T \in \hom(V,W)\) então, \(\psi = T \circ \phi \in  \hom^k(\Pi V_j, W)\). Pra ver a linearidade mostra que \(\forall \psi, \xi \in \hom^k(\Pi V_j, W)\) e \(\forall \lambda \in \F\), \((\widetilde{\psi}+ \lambda \widetilde{\xi}) \circ \phi = \psi + \lambda \xi\). } 

\begin{note}
    Notação padrão pra producto tensorial, \(\textstyle (\phi, V) = \left(\otimes, \otimes V_j\right)\). A $k$-linearidade fica,  \( \cdots \otimes v_{1,j_0} + \lambda v_{2,j_0} \otimes \cdots = \otimes ((v_{1,j})_{j_0}) + \lambda \otimes ((v_{2,j})_{j_0})\).
    %O corpo e \(k\) apareceram só se precisar.
%    \[\otimes ((v_{1,j})_{j_0} + \lambda(v_{2,j})_{j_0}) = \otimes (v_{1,j})_{j_0} + \lambda \otimes (v_{2,j})_{j_0}.\]
\end{note}

\propositionnum{10.2.3.}
\begin{proposition}
    \label{prop:1023}
    Dado \(1\leq l < k\), \(\exists ! \Gamma : \otimes V_j \to (\otimes^l V_j) \otimes (\otimes_{l+1}^k  V_j)\)  isomorfismo tal que, \(\forall j\leq k\), \(\forall v_j \in V_j\), \(\Gamma:  \otimes v_j \mapsto (\otimes^l v_j) \otimes (\otimes_{l+1}^kv_j)\). 
\end{proposition}

\demo{Defina \(\phi_b : \Pi V_j \to W =  (\otimes^l V_j) \otimes (\otimes^k_{l+1} V_j)\) e tente mostrar que o par \((\phi_b, W)\) também é universal em \(\Pi V_j\). O resultado segue do {\scriptsize\(\square\) }\hyperlink{lemma1013}{10.1.3.}.}

\propositionnum{10.2.4.}
\begin{proposition}
    \(\forall j\leq k\) sejam \(W_j \leq V_j\). Então, \(\exists ! \tilde{\Gamma} : \otimes W_j \to \otimes V_j\) linear injetora tal que \(\forall j\leq k, \ \forall w_j \in W_j, \ \tilde{\Gamma} : \otimes w_j \mapsto \otimes w_j\). 
\end{proposition}

\demo{No contexto da \(\square\) \ref{prop:1023} \(\exists ! \Gamma_{_V} :\otimes V_j \to \otimes (V_j)\) e \(\exists !\Gamma_{_W}: \otimes W_j \to \otimes (W_j)\) isomorfismos. Faça \(\tilde{\Gamma} = \Gamma^{-1}_{_V}\circ \Gamma_{_W} \). }

\begin{corollary}
    \(\otimes v_j = 0 \leftrightharpoons \exists j\leq k : v_j = 0 \). 
\end{corollary}

\begin{note}
    Do \(\blacksquare\) \ref{thm:1021} sabemos que \([\otimes v_j] = \otimes V_j\), então \(\forall \textbf{v} \in \otimes V_j,\ \exists m \in \mathbb{Z}_{\geq 0}\) e \( \exists v_{i,j} \in V_j \setminus\{0\}: \textbf{v} = \sum_{i\leq m} \otimes v_{i,j} \), ou seja, \textbf{v} é soma de \emph{tensores puros}.   
\end{note}

\begin{definition}
    Se \(\textbf{v} \in \otimes V_j\) então \(\pt(\textbf{v}) := \min\{m \in \mathbb{Z}_{\geq 0} : \textbf{v} = \sum_{i\leq m} \otimes v_{i,j}\}\). 
\end{definition}
