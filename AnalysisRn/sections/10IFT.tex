\subsection*{Teorema da Função Inversa}

\begin{definition}
    \(F : U\ab\subseteq \R^n\to \R^n\) é \emph{diffeomorfismo local} de classe \(C^k\) em \(p\in U\) se \(\exists \mathcal{V}_p\subseteq U\) tal que \(^k F: \mathcal{V}_p\simeq F(\mathcal{V}_p)\ab \subseteq \R^n\). Denotamos, \(^k F: \mathcal{V}_p\). 
\end{definition}

%\begin{example}
%    \(\forall \theta \in  \R\) la aplicación \(F: (r,\theta)\mapsto (r\cos(\theta),r\sin(\theta))\) es \(^k F: \mathcal{V}_{(r,\theta)}\). 
%\end{example}

\begin{theorem}[\emph{Função Inversa}]
\hypertarget{thmTFI}{}
    Sejam \(U\ab \subseteq \R^n\), \(F \in C^k(U, \R^n)\) e \(p\in U\). Então, a aplicação  é \(^k F : \mathcal{V}_p\) \(\leftrightharpoons\) \(DF(p)\) é isomorfismo.  
\end{theorem}

\begin{definition}

\end{definition}

\hypertarget{thmPFB}{}
\begin{theorem}[\emph{Ponto Fixo Banach}]
    Se \(X\ce\subseteq \R^n\) e \(T:X\to X\) é uma contração, então \(\exists! \ x^*\in X \) tal que \(T(x^*)= x^*\) e \(\forall x_0 \in X, \ \lim\limits_{n\to \infty} T^n(x_0) = x^*\).   
\end{theorem}

\begin{corollary}
    \textbf{8.4.} Seja \(F \in C^k(U,\R^n)\) tal que \(\forall p \in U\), \(\det JF(p)\neq 0\), então \(F\) é uma aplicação aberta. Mas ainda, se \(F\) for injetora, então \(^k F: U \simeq F(U)\). 
\end{corollary}

\hypertarget{prop85}{}
\begin{proposition}
    \textbf{8.7.} Sejam \(^k H\subseteq \R^n\) hipersuperfície definida por \(g(\textbf{x})=0\), \(p\in H\) e \(c\in C^k(\mathcal{V}_\epsilon,H)\) um caminho em \(H\). Então, \(\ker dg(p)= \{\dot{c}(0): c(0) = p \}\).
\end{proposition}

\demo{(\(\supseteq \)) Use \(\blacksquare\) \hyperlink{thmRDC}{(\emph{Regra da Cadeia})} na função \((g \circ c)(t)\). (\(\subseteq\)) \(\exists i \leq n \) tal que \(g_{x_i}(p) \neq 0\), suponha que é a última \(i=n\). Tome \(\vec{v}\in \R^n: dg(p)\cdot v=0\), aplique \(\blacksquare\) \hyperlink{thmTFIm}{(\emph{Função Implícita})}, assim, \(c(t)= (p_j + tv_j, f(p_j+tv_j))\). Finalmente, use nele o {\scriptsize \(\boxtimes\)} \hyperlink{corol83}{8.5.}. }

\begin{definition}
    Seja \(p \in \ ^k H\subseteq \R^n\) hipersuperfície. O \emph{espaço tangente} a \(H\) em \(p\) é    
    \[T_pH:= \ker dg(p) = \{\vec{v} \in \R^n: \langle \nabla g(p), v\rangle = 0\}. \footnote{\(T_pH\leq \R^n\) não necessariamente passa por \(p\).}\]
\end{definition}

\end{example}
