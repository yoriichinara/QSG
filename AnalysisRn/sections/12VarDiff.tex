\section{Variedades Differenciáveis}

\begin{definition}
    Sejam \(X\subseteq \R^n\) e \(Y\subseteq \R^m\). Uma aplicação \(F\in C^k(X,Y)\) sse \(\forall p \in X\), \(\exists \widetilde{F} \in C^k(\U_p, \R^m)\) tal que \( \widetilde{F}|_{\U_p \cap X} = F\).    
\end{definition}

%\begin{note}
%    Análogamente, se \(F \in C^\infty(X;Y)\), então dizemos que é \emph{suave}.
%\end{note}

\begin{definition}
    \(F: X \to Y\) é \emph{diffeomorfismo} de classe \(C^k\) se \(F\in C^k(X,Y)\) é homeomorfismo e \(F^{-1}\in C^k(Y,X)\). Denotamos \(^k F : X \simeq Y \). 
\end{definition}
\begin{example}
    Sejam \(X= \{(x,y) \in \mathbb{S}^1: y > 0 \}\) e \(Y = (-1,1)\). A aplicação \(\pi: X \ni (x,y) \mapsto x \in Y\) é \(^\infty F: X\simeq Y \). \textcolor{gray}{\(\rightarrow\) Verifique as condições}
\end{example}
\begin{definition}
    Um subconjunto \(^k_d M \subseteq \R^n\) é uma \emph{subvariedade} de classe \(C^k\) e dimensão \(d\), se  \(\forall p \in M \), \(\exists\ ^k \varphi: \U_p \cap M \simeq V\ab \subseteq \R^d\) \emph{carta local}. As componentes de \(\varphi = (\x_1, \ldots, \x_d)\) são \emph{coordenadas locais} e \(\psi := \varphi^{-1} : V \to \U_p \cap M \) é uma \emph{parametrização local}.
    %\begin{itemize}[label = \(\star\)]
    %    \item \(\varphi = (\x_1, \ldots, \x_d)\) \(\leftarrow \) \emph{Coordenadas locais}.
    %    \item \(\psi := \varphi^{-1} : V \to \U_p \cap M \) \(\leftarrow\) \emph{Parametrização local}.
    %\end{itemize}
\end{definition}

\begin{note}
    Daqui pra frente nós entendemos \(\U_p\) vizinhança de \(p\in M\leq \R^n\). 
\end{note}

\begin{example}
    \(^{\infty}_{\hspace{0.15cm}n} U\ab\subseteq \R^n\) e \(^{\hspace{0.3cm}\infty}_{n-1}\E^{n-1} \subseteq \R^n\). \comentario{ \(\varphi = \psi = \id_n\) e \(\varphi = \pi_{_N}\), respectivamente }
\end{example}

\begin{lemma}
    Sejam \(^k_d M \subseteq \R^n \) e \(\psi: V\ab\subseteq \R^d \to M\subseteq \R^n\). Então, \(\psi\) é parametrização local da \(M\) \(\leftrightharpoons\) \(\psi \) é imersão e homeomorfismo \((M,\tau_{_M})\). 
\end{lemma}

\demo{(\(\Rightarrow\)) Direta da definição. (\(\Leftarrow\)) A bijeção é imediata de \(\psi(V)\ab \subseteq M\). Aplique \(\blacksquare\) \hyperlink{thmFLI}{(\emph{FLI})} e faça composição com a projeção. }

\begin{proposition}
    \(F\in C^k(M,N)\) \(\leftrightharpoons\) \(\forall \psi: V\ab \to \U_p\) parametrização local da \(M\) em \(p\), tem-se que \(F \circ \psi \in C^k(V, N)\). \textcolor{gray}{\(\rightarrow\) Basta ver só um  atlas}  
\end{proposition}

\begin{definition}
    Seja \(F\in C^k(M,N)\). A derivada de \(F\) em \(p \in M\) é a aplicação linear \(DF(p): T_pM \to T_{F(p)}N \) tal que \( v\mapsto D\widetilde{F}(p) \cdot v\). 
\end{definition}

\begin{lemma}
    Se \(F\in C^k(M,N)\), então \(DF(p)\) não depende da \(\widetilde{F}\).  
\end{lemma}

\subsection*{Valores Regulares}

\begin{definition}
    Seja \(F \in C^k(M,N)\). Um ponto \(p\in M\) é \emph{regular} se \(F\) é submersão em \(p\), se não for regular, então é \emph{ponto crítico} de \(F\).  
\end{definition}

\begin{note}
    Se \(\exists p\in M\) ponto regular de \(F\), então \(\dim M \geq \dim N\).
\end{note}

\ejemplo{ \centering \(\blacksquare\) \hyperlink{thmML}{(\emph{Multiplicadores de Lagrange})}
\tcblower
\begin{example}
    Sejam \(M= \ ^k H\subseteq \R^n\) hipersuperfície definida por \(g(\x)=0\), \(U\ab\subseteq \R^n: H\subset U\) e \(f: U \to \R\). Então, \(p \in M\) é crítico \(\textcolor{verdeoscuro}{\leftrightharpoons}\  df(p): T_pM \not \twoheadrightarrow \R\).
\end{example}
}

%\{El multiplicador de Lagrange, de nuevo }

\begin{definition}
    Sejam \(F\in C^k(M,N)\) e \(q\in N\). Dizemos que \(q\) é \emph{valor crítico} se \(\exists p\in M\) tal que \(p \) é ponto crítico e \(F(p)= q\), de outra forma é \emph{valor regular} da \(F\). Conjuntos da forma \(F^{-1}(q)\) são chamados de \emph{fibras}.
\end{definition}

\alerta{ \centering \(p\in M\) regular \textcolor{red}{\(\nRightarrow\)} \(q = F(p)\) regular}

\begin{theorem}[\emph{Fibras Regulares}]
    Sejam \(^{\hspace{0.1cm}k}_{d_1} M\subseteq \R^n\), \(^{\hspace{0.1cm}k}_{d_2}N\), \(F\in C^k(M,N)\). Se \(q \in N\) é valor regular da \(F\), então \(F^{-1}(q) =: \ ^{\hspace{0.85cm}k} _{d_1 - d_2}P \subseteq M  \subseteq \R^n\).
\end{theorem}

\demo{Sejam \(p = F^{-1}(q)\) e \(d_1 - d_2 = c \). Note que \(\R^c \cong \ker DF(p) \leq \R^n\), defina \(L: \R^n \to \R^c\) tal que \(L\cdot \ker DF(p)\) é isomorfismo. Faça \(G:M \ni \x\mapsto (F(\x), L(\x)) \in  N \times \R^c\), use \(\blacksquare\) \hyperlink{thmTFIV2}{(\emph{Função Inversa V2})} e conclua. }

\begin{example}
    \(1\) é valor regular de \(g: \R^n \ni \textbf{x}\mapsto  \| \x \|^2 \in \R\) e \(g^{-1}(1)= \E^{n-1}\). 
\end{example}

\ejemplo{ \centering Variedades de dimensão \(0\) 
\tcblower 
\begin{example}
    Sejam \(F\in C^k(M,N)\) com \(\dim M = \dim N \) e \(q \in N \) valor regular de \(F\). Então, \(P = F^{-1}(q)\) é um conjunto discreto. Mas ainda, se \(M\) for compacta, então \( |P| < \infty\). \textcolor{gray}{\(\rightarrow \ \R^0 = \{\text{o}\} \) }
\end{example}
}

\begin{theorem}[\emph{Brown-Sard}]
    Seja \(F\in C^k(M,N)\). Se \(k \gg 0\), então o subconjunto de valores regulares de \(F\) é denso em \(N\).% \footnote{Se exige \(k\geq \dim M - \dim N + 1\) pero así se ve mas bonito }
\end{theorem}

\begin{note}
    A \emph{medida} de \(B= (a_j,b_j)^n\subseteq \R^n\) bloco aberto é \(\mu(B):= \prod (b_j-a_j)\).
\end{note}

\begin{definition}
    \(S\subseteq \R^n\) tem \emph{medida nula} sse \(\forall \epsilon >0\),  \(\exists  (B_i)\) familia de blocos abertos tais que  \(S\subseteq \bigcup B_i\) e \(\sum \mu(B_i)< \epsilon\). .  
\end{definition}

\begin{example}
    \(S\subseteq \R^n\) enumerável tem medida nula.  \textcolor{gray}{\(\rightarrow\) São pontos isolados} 
\end{example}
\hypertarget{thmSard}{}
\begin{theorem}[\emph{Sard}]
    Sejam \(U\ab\subseteq \R^n\) e \(F\in C^k(U, \R^m)\) tal que \(k\geq \{n-m+1, 1\}\). Então, o conjunto dos valores críticos de \(F\) tem medida nula em \(\R^m\). 
\end{theorem}

\begin{corollary}
    %Segue do \(\blacksquare\) \hyperlink{thmSard}{(\emph{Sard})} que  
    Se \(d<n\), então \(^k_d M\subseteq \R^n\) tem medida nula. \textcolor{gray}{\(\rightarrow\) Todos os valores da inclusão \(\iota : M \hookrightarrow \R^n\) são críticos}
\end{corollary}

\begin{exercise}
    Se \(S\subseteq \R^n \) tem medida nula, então \(\R^n\setminus S \) é denso em \(\R^n\). 
\end{exercise}
