Nesta seção \(I := [0,1]\subset \R\). Até agora usamos \(\textcolor{gray}{(-)}'\) únicamente para o pull-back \(\sim\) espaço dual, isso vai continuar assim, não confundir com derivadas.  

\subsection*{Integração em Cadeias}

\begin{definition}
    Um \(k\)\emph{-bloco} em \(M\subseteq \R^n\) é uma aplicação \(\sigma \in C^\infty(\I^k, M)\).
\end{definition}

%\ejemplo{ \centering  
%    \tcblower
\begin{example}
  \(\id_{\I^n} : \I^n \to \I^n \subset \R^n \), $n$\emph{-bloco padrão}. As curvas são  \(1\)-blocos. 
\end{example}

\alerta{
\centering \(\sigma \  \ k\)-bloco \(\textcolor{red}{\neq}  \ \ \operatorname{Im}\sigma\subseteq M\) 
\tcblower
\begin{example}
    \(\sigma: \I^n \ni \x \mapsto p \in M \) é um $n$-bloco.%, chamamos ele em particular de $n$\emph{-cubo singular}. 
\end{example}
}

\begin{definition}
    Sejam \(\omega \in \Omega^k(M)\) e \(\sigma \) um $k$-bloco. Então,   
    \[\int_\sigma \omega := \int_{\I^k} \sigma'(\omega) = \int_{\I^k}f(\x) \ dx_1\wedge \cdots \wedge dx_k =\int_{\I^k} f(\x) \ d\x. \]  
\end{definition}

\ejemplo{
\centering Integral de linha
\tcblower
\begin{example}
    Sejam \(c(t) : [a,b] \to U\ab\subseteq \R^n\) curva suave e \(\sum^n f_j \ dx_j = \omega \in \Omega^k(U) \), então \(\oint_c \omega = \int_a^b c'(\omega) \ dt = \int_a^b \sum^n (f_j \circ c)(t) \dot{c_j}(t) \ dt\ .\)
\end{example}
}

\begin{definition}
    Sejam \(\omega \in \Omega^k(M)\) e \(c \) $k$-cadeia, então \(\displaystyle \int_c \omega :=  \sum^k n_i\int_{\sigma_i} \omega .\)  
\end{definition}

\begin{definition}
    Seja \(\sigma \) um $k$-bloco. O \emph{borde} de \(\sigma\) é a \((k-1)\)-cadeia dada por
    \[\hspace{5cm} \partial \sigma = \sum^k (-1)^i (\sigma_{(i,0)}- \sigma_{(i,1)}).\]
\end{definition}

\begin{exercise}
    Se \(c\) é uma $k$-cadeia, então \(\partial (\partial c) = 0 \). \textcolor{gray}{\(\rightarrow\) Olha no \cite{spivak2018}}  
\end{exercise}

\hypertarget{thmStokes}{}
\begin{theorem}[\emph{Stokes}]
    Sejam \(\omega \in \Omega^{k-1}(M)\) e \(c\) $k$-cadeia em \(M\). Então, \(\displaystyle \int_c d\omega = \int_{\partial c } \omega\).
\end{theorem}

\demo{
  \textcolor{black}{\emph{Nota.}} Podemos fazer algumas suposições inicias: \textcolor{black}{
\begin{enumerate}[label = \roman*., left = -0.5cm]
    \item \(c = \sigma \) $k$-bloco. \comentario{Se valer \(\blacksquare\) \hyperlink{thmStokes}{(\emph{Stokes})} pra $k$-blocos, então \(\int_c d\omega = \sum n_i \int_{\sigma_i} d \omega = \sum n_i \int_{\partial \sigma_i} \omega = \int_{\sum n_i \partial \sigma_i} w = \int_{\partial c} w \) }
    \item \(\sigma = \id_{\I^k} \) e \(M\) uma vizinhanza de \( \I^k\). \comentario{\(\textstyle\int_\sigma d\omega = \int_{\id_{\I^k}} \sigma ' (dw) = \int_{\id_{\I^k}} d (\sigma '(w)) = \int_{\partial \id_{\I^k}} \sigma '(w) = \int_{\partial \sigma} w \). }
    \item \(\omega = f \hat{w}_i \in \Omega^{k-1}(U),\ U\ab\subseteq \R^k : \I^k\subset U\). \comentario{\(\textstyle\int_{\sigma} d\omega = \int_{\sigma} d\left(\sum f_i \hat{\omega}_i\right) = \sum \int_{\sigma} d\left( f_i \hat{\omega}_i\right) = \sum \int_{\partial \sigma}  f_i \hat{\omega}_i = \int_{\partial \sigma} \omega\). }
\end{enumerate}}
}

\subsection*{Integração em Variedades com Bordo}

\begin{definition}
  \centering \(\mathbb{H}^n := \{(x_1, \ldots, x_n) \in \R^n : x_n \geq 0\}\). 
\end{definition}

\begin{definition}
  Seja \(_dM\subseteq \R^n\) variedade. O subconjunto \(_d\overline{M}\subseteq  \R^n\) é uma \emph{variedade com bordo} \( \overline{M}\setminus M =:\  _{d-1}\partial M\subseteq \R^n  \). Mas agora temos, \(\forall p \in \overline{M}, \ \exists ^k\varphi : \U_p \cap \overline{M} \simeq V\ab\subseteq \mathbb{H}^d\). 
\end{definition}

%\begin{note}
%  \(\partial M = \{p \in M \ | \ \exists ^k\varphi  : \varphi(p) = (x_1, \ldots, x_d= 0)\}\). 
%\end{note}

\begin{theorem}
  Sejam \(_dM\subseteq \R^n\) e \(g \in C^\infty(M)\). Se \(0\) é valor regular de \(g\), então \(\overline{N}:= g^{-1}([0,\infty)) \) é \(_d\overline{N} \subseteq \R^n\) e \(\partial N = g^{-1}(0)\). \comentario{Usa \(\blacksquare\) \hyperlink{thmFLS}{(\emph{FLS})} em \( \{0\}\cup (0,\infty)\) }
\end{theorem}

\end{note} 

\begin{definition}
  Sejam \(_d\overline{N}\subseteq \R^n\) e \(\omega \in \Omega^d(\overline{N})\). Chamamos de \emph{orientación inducida} por \(\omega \) à \(\Omega ^d(\partial N) \ni \eta : (v_1, \ldots, v_{d-1}) \mapsto \omega(\nu(p), v_1, \ldots, v_{d-1})\).  
\end{definition}

\begin{example}
  Em \(\overline{\mathbb{D}} \subseteq  \R^2\), \(\omega = dx \wedge dy\) temos \(\eta (p)(v) = \det(p,v) = d\theta |_{_{\E^1}}\). 
\end{example}
