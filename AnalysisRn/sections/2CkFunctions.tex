\section{Funções de Classe \(C^k\)}
 
\begin{definition}
    Seja \(f: U\ab\subseteq \R^n \to \R\) tal que \(\forall p \in U, \exists \frac{\partial f }{\partial x_j}(p)\), então sendo \(\frac{\partial f }{\partial x_j}: U \to \R\) função, uma \emph{derivada de segundo ordem} de \(f\) é dada por   
%    sendo  bem definida,
    \[\frac{\partial^2 f}{\partial x_i \partial x_j}(p) := \frac{\partial}{\partial x_i}\left(\frac{\partial f}{\partial x_j}\right)(p).\footnote{O ordem importa em geral. Analogamente são definidas as $k$-ésimas derivadas  de \(f\).}\]
\end{definition}

\begin{definition}
    Seja \(k\in \Z_{\geq 0} \). Uma função \(f:U\ab\subseteq \R^n\to \R\) é de classe \(C^k(U)\) sse \(\forall |\alpha| \leq k, \exists \partial^\alpha f: U \to \R\) e são continuas. 
\end{definition}

\begin{note}
\(f \in C^0\leftrightharpoons f\) continua; \(f \in C^\infty \leftrightharpoons f\) é \emph{suave}.  
\end{note}
%Note que van a existir varias derivadas de orden \(k\), dependiendo del valor de \(n\)
\begin{example}
    O anel de polinômios \(\R[x_1,\ldots,x_n]\subseteq C^\infty(\R^n)\). Portanto também, \(\det \in C^\infty\left(\R^{n^2}\right)\). \textcolor{gray}{\(\rightarrow\) A formúla de Laplace diz que é polinômial}
\end{example}
\begin{example}
    Seja \(f: x\mapsto x^{1/3}\), então \(f \in C^0(\R)\setminus C^1(\R)\) pois \(\nexists \dot{f}(0)\). \textcolor{gray}{\(\rightarrow\) Veja o que acontece com o límite nesse punto}
\end{example}

\begin{theorem}[\emph{Schwarz}] 
\hypertarget{thmSchw}{}
    Se \(f \in C^2(U)\), então \(\forall i,j \leq n\), \ \(\displaystyle\frac{\partial^2 f}{\partial x_i \partial x_j} = \frac{\partial^2 f }{\partial x_j \partial x_i}\). 
\end{theorem}
