\section{Funções de Classe \(C^k\)}
 
\begin{definition}
    Sejam \(j\leq n\) e  \(f: U\ab\subseteq \R^n \to \R\) tal que \(\forall p \in U, \exists f_{x_j}(p)\). Então, sendo \(f_{x_j}: U \to \R\) função, uma \emph{derivada de segundo ordem} de \(f\) é dada por   
%    sendo  bem definida,
    \[(f_{x_j})_{x_i}(p) = \frac{\partial^2 f}{\partial x_i \partial x_j}(p) := \frac{\partial}{\partial x_i}\left(\frac{\partial f}{\partial x_j}\right)(p).\footnote{O ordem importa em geral. Analogamente são definidas as $k$-ésimas derivadas  de \(f\).}\]
\end{definition}


\begin{definition}
    \vspace{-0.5cm}Seja \(k\in \Z_{\geq 0} \). Uma função \(f:U\ab\subseteq \R^n\to \R\) é de classe \(C^k(U)\) sse \(\forall |\alpha| \leq k, \exists \partial^\alpha f: U \to \R\) e são contínuas. 
\end{definition}

\begin{note}
\(f \in C^0\leftrightharpoons f\) contínua; \(f \in C^\infty \leftrightharpoons f\) é \emph{suave}.  
\end{note}
%Note que van a existir varias derivadas de orden \(k\), dependiendo del valor de \(n\)
\begin{example}
    \(\mathcal{P}[x_1,\ldots,x_n]\subseteq C^\infty(\R^n)\). Em particular, \(\det \in C^\infty\left(\R^{n^2}\right)\). \textcolor{gray}{\(\rightarrow\) Pela fórmula de Laplace o determinante é polinomial}
\end{example}
\begin{example}
    Seja \(f: x\mapsto x^{1/3}\), então \(f \in C^0(\R)\setminus C^1(\R)\) pois \(\nexists \dot{f}(0)\). \textcolor{gray}{\(\rightarrow\) Veja o que acontece com o limite nesse ponto}
\end{example}

\begin{theorem}[\emph{Schwarz}] 
\hypertarget{thmSchw}{}
    Se \(f \in C^2(U)\), então \(\forall i,j \leq n\), \ \(\displaystyle\frac{\partial^2 f}{\partial x_i \partial x_j} = \frac{\partial^2 f }{\partial x_j \partial x_i}\). 
\end{theorem}

\begin{corollary}
    Se \(f\in C^k(U)\), então não importa a ordem em que são tomadas as derivadas de ordem \(|\alpha|\leq k\). \textcolor{gray}{\(\rightarrow\) É questão de trabalhá-las 2 a 2}
\end{corollary}
