O objetivo nesta seção é geralizar o conceito de integral, reversõando todos os resultados e propiedades que a gente tinha lá em Cálculo I.

\subsection*{Formas Alternadas}

\begin{definition}
    Uma $k$-forma \(\phi \in \hom^k_\R(V, \R )\) é \emph{alternada} sse  toda vez que \(\exists i<j\)  tal que \(v_i =v_j\) temos que \(\phi((v_{i<j})):= \phi(v_1, \ldots, v_i, \ldots, v_j, \ldots, v_k) = 0\). 
\end{definition}

%\end{example}

\hypertarget{lemma101}{}
\begin{lemma}
    \textbf{10.1.} Sejam \(\phi \in \operatorname{Alt}^k_\R(V)\) e \(i<j\). Então,  
    \begin{enumerate}[label = \roman*.]
        \item \(\phi ((v_{i<j})) = -\phi ((v_{j<i}))\).
        \textcolor{gray}{\(\rightarrow \) Faça a conta \(\phi((v_{i<j}) + v_j \cdot e_i + v_i\cdot e_j)\)}
        \item Se \(v_1, \ldots, v_k\) foram l.d. então \(\phi ((v_j)) = 0\). \textcolor{gray}{\(\rightarrow\)  \(\exists j_0\leq k : v_{j_0} =  \sum_{j\neq j_0} a_jv_j\), separa usando linearidade em \(j_0\)}
    \end{enumerate}
\end{lemma}

\begin{corollary}
    As $k$-formas alternadas são antisimétricas. Se \(S_k\) é o grupo de permutações de ordem \(k\) e \(\operatorname{sgn} : S_k  \to \{-1, 1\}\) então, \(\phi((v_{\sigma(j)})) = \operatorname{sgn}(\sigma) \cdot \phi((v_j))\). \footnote{\(S_k \ni \sigma = \prod \tau_m : \tau_m\) é transposição, logo, \(\operatorname{sgn}(\sigma) = (-1)^{\#M}\). }
\end{corollary}

\begin{exercise}[\emph{Projetor Alternado}]

\end{exercise}

\begin{definition}
    Sejam \(T \in \hom_\R(V,W)\) e \(\phi \in \operatorname{Alt}^k_\R(W)\). O \emph{pullback} de \(\phi \) por \(T\) é a função \(T'(\phi) \in \operatorname{Alt}^k_\R(V)\) tal que \((v_j) \mapsto \phi ((T (v_j)))\). \footnote{Da uma olhada nas Notas de Algebra Linear II, Seç. 9.2..}
\end{definition}

\begin{definition}
    Sejam \(\phi \in \operatorname{Alt}^p_\R(V)\) e \(\psi \in \operatorname{Alt}^q_\R(V)\). O \emph{produto exterior} de \(\phi\) e \(\psi\) é a função \(\phi \wedge \psi \in \operatorname{Alt}^{p+q}_\R (V)\) dada por,  
    {\[(v_1, \ldots, v_{p+q}) \mapsto \frac{1}{p!q!} \sum_\sigma \operatorname{sgn}(\sigma) \phi(v_{\sigma(1)}, \ldots, v_{\sigma(p)}) \psi(v_{\sigma(p+1)},\ldots, v_{\sigma(p+q)}).\]}
\end{definition}

\begin{example}
    Sejam \(\phi, \psi \in \operatorname{Alt}^1_\R(V)\). Então,  \(\phi \wedge \psi \in \operatorname{Alt}^2_\R(V)\) é dada por,
    \vspace{-0.2cm}
    \[(v_1, v_2) \mapsto \phi(v_1)\psi(v_2) - \phi(v_2)\psi(v_1) =\det  \begin{bmatrix}
        \phi(v_1) & \psi(v_1) \\ 
        \phi(v_2) & \psi(v_2) 
    \end{bmatrix}.\] 
\end{example}

\begin{proposition}
    Sejam \(\phi \in \alt^p(V),\ \psi \in \alt^q(V)\) e \(\xi \in \alt^r(V)\). Então, o produto \(\wedge\) é:   
    \begin{enumerate}[label = \roman*.]
        \item Bém definido, \(\wedge:  (\phi, \psi )\mapsto \phi \wedge \psi \in \alt^{p+q}(V)\).   
        \item Bilinear, se \(p = r \Rightarrow (\phi + \lambda \xi) \wedge \psi =  \phi \wedge \psi + \lambda (\xi \wedge \psi)\), na outra coordenada, se \(q = r \Rightarrow \phi  \wedge (\psi+ \lambda \xi) =  \phi \wedge \psi + \lambda (\phi \wedge \xi) \). %\textcolor{gray}{\(\rightarrow \) ii. e v. Directos de la definición} 
        \item Anticonmutativo, \(\phi \wedge \psi =  (-1)^{pq} \psi \wedge \phi\). \textcolor{gray}{\(\rightarrow \) Tome a permutação \[\begin{pmatrix}
    1\ \ \ \cdots \ \ \  p & p+1  \cdots  p+q\\ 
    q+ 1 \cdots  q+p & 1 \ \ \  \cdots \ \ \  q
\end{pmatrix}\] }
\vspace{-0.6cm}
        \item Associativo, \((\phi \wedge \psi) \wedge \xi = \phi \wedge (\psi \wedge \xi)\). \textcolor{gray}{\(\rightarrow \) Use o projetor alternado em \(f(v_i), \ i \in \{n\in \N : n\leq p+q+r\}\) e faça as contas chatas}
        \item Compativél com pullback, \(T'(\phi \wedge \psi) = T'(\phi) \wedge T'(\psi)\).   
    \end{enumerate} 
\end{proposition}

\subsection*{Formas Differenciais}

\begin{note}
    Se \( \varphi = (\x_1, \ldots, \x_d): \U_p \to \R^d\) é uma carta local de \(^r_dM \subseteq \R^n\), lembre-se que \([d\x_1, \ldots, d\x_d]  = (T_pM)' = \alt^1(T_pM)\). %Também, \(A^k(T_pM) = [d\x_{i_1}\wedge \cdots \wedge d\x_{i_k}]\). 
\end{note}
\- \vspace{-0.8cm}
\begin{definition}
    Uma \emph{$k$-forma diff} de classe \(C^r\) em \(^r_dM\subseteq \R^n\), é uma associação 
    \[^r_k\omega : M \ni p \mapsto \omega(p) = \sum f_{i_1, \ldots, i_k} \ d\x_{i_1}\wedge \cdots \wedge d\x_{i_k} \in \alt^k(T_pM), \] 
    tal que, \(\forall \varphi = (\x_1, \ldots, \x_d)\), as funções \(f_{i_1, \ldots, i_k}\in C^r ( \U_p,\R)\).
\end{definition}

\subsection*{Derivada Exterior}

Agora precisamos entender como é que se derivam as formas diferenciais. 

\hypertarget{thm102}{}
\begin{theorem}
    \textbf{10.2.} \(\forall M\subseteq \R^n, \ \forall k \in \N\), \(\exists !(d_j) \in \hom_\R(\Omega^k(M), \Omega^{k+1}(M))\), familia de operadores tais que \(d_j: \omega \mapsto d\omega\) verifica:  
    \begin{enumerate}[label= \roman*.]
        \item Derivada de uma função, \(\Omega^0(M) = C^\infty(M) \ni f \mapsto df \in \Omega^1(M)\). 
        \item Regla de Leibniz com sinal, se \(\omega \in \Omega^p(M)\) e \(\eta\in \Omega^q(M)\) então, 
        \[ d(\omega \wedge \eta ) = d\omega \wedge \eta + (-1)^p \omega \wedge d \eta. \] \vspace{-1cm}
        \item \(d^2 =0\), se \(\omega \in \Omega^k(M)\) então \(d (d\omega) = 0\). \textcolor{gray}{\(\rightarrow \) Vem do \(\blacksquare\) \hyperlink{thmSchw}{(\emph{Schwarz})}}
        \item Compativél com pullback, se \(F \in C^\infty (M,N)\) e \(\omega \in \Omega^k(N)\), então temos \(d(F'(\omega)) = F'(d\omega)\). 
    \end{enumerate}
\end{theorem}

\begin{definition}
    \(\omega \in \Omega^k(M)\) é \emph{exata} se \(\exists \eta \in \Omega^{k-1}(M)\) tal que \(\omega = d\eta\). Por outro lado, se \(d\omega = 0 \) dizemos então que \(\omega\) é \emph{fechada}. 
\end{definition}
\begin{note}
    Ver que \(\omega \) é exata é "resolver" EDPs, ou seja integrar. %, por isso também que escrever \(f_{x_j}= \frac{\partial f}{\partial x_j}\) parece mais conveniente agora. 
\end{note}
